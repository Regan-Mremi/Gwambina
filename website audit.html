3.1. Server Hardening & Nginx Production Configuration

This configuration enforces HTTPS, configures TLS 1.2/1.3, implements robust security headers, enables Gzip compression, and manages canonical HTTP-to-HTTPS redirects.

/etc/nginx/sites-available/gwambina.co.tz.conf
# Nginx Server Configuration for www.gwambina.co.tz
# Handles HTTPS redirection, security headers, caching, and compression

# Redirect HTTP to HTTPS (Both naked and www)
server {
listen 80;
listen [::]:80;
server_name gwambina.co.tz www.gwambina.co.tz;

location /.well-known/acme-challenge/ {
root /var/www/html;
}

location / {
return 301 https://www.gwambina.co.tz$request_uri;
}
}

# Redirect HTTPS naked domain to HTTPS www
server {
listen 443 ssl http2;
listen [::]:443 ssl http2;
server_name gwambina.co.tz;

ssl_certificate /etc/letsencrypt/live/gwambina.co.tz/fullchain.pem;
ssl_certificate_key /etc/letsencrypt/live/gwambina.co.tz/privkey.pem;

return 301 https://www.gwambina.co.tz$request_uri;

}

# Primary HTTPS Server Block
server {
listen 443 ssl http2;
listen [::]:443 ssl http2;
server_name www.gwambina.co.tz;

root /var/www/gwambina_html;
index index.html index.php;

# SSL Certificates (Let's Encrypt / Certbot)
ssl_certificate /etc/letsencrypt/live/gwambina.co.tz/fullchain.pem;
ssl_certificate_key /etc/letsencrypt/live/gwambina.co.tz/privkey.pem;
ssl_dhparam /etc/ssl/certs/dhparam.pem;

# Modern SSL Protocol & Cipher Tuning
ssl_protocols TLSv1.2 TLSv1.3;
ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-
AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384;
ssl_prefer_server_ciphers off;
ssl_session_timeout 1d;
ssl_session_cache shared:SSL:10m;
ssl_stapling on;
ssl_stapling_verify on;

# Robust Security Headers
add_header X-Frame-Options "SAMEORIGIN" always;
add_header X-XSS-Protection "1; mode=block" always;
add_header X-Content-Type-Options "nosniff" always;

access_log off;

location / {

}

# Deny access to hidden files (.htaccess, .git, .env)
location ~ /\. {
deny all;
access_log off;
log_not_found off;
}
}

3.2. Automated SSL Provisioning Script

Bash script to automatically install Certbot, obtain Let's Encrypt certificates for gwambina.co.tz, and set up automatic renewal cron jobs.

setup_ssl.sh
#!/usr/bin/env bash
# Automated SSL Provisioning & Renewal Script for gwambina.co.tz
set -e

DOMAIN="gwambina.co.tz"
WWW_DOMAIN="www.gwambina.co.tz"
EMAIL="admin@gwambina.co.tz"

echo "=== Updating System & Installing Certbot ==="
apt-get update -y
apt-get install -y certbot python3-certbot-nginx openssl

echo "=== Generating Strong Diffie-Hellman Group ==="
if [ ! -f /etc/ssl/certs/dhparam.pem ]; then
openssl dhparam -out /etc/ssl/certs/dhparam.pem 2048
fi

echo "=== Requesting Let's Encrypt SSL Certificate ==="
certbot certonly --nginx \
--non-interactive \
--agree-tos \
--email "$EMAIL" \
-d "$DOMAIN" -d "$WWW_DOMAIN"

echo "=== Testing Renewal Process ==="

certbot renew --dry-run

echo "=== Setting up Automated Renewal Cron Job ==="
(crontab -l 2>/dev/null; echo "0 3 * * * /usr/bin/certbot renew --quiet --post-hook 'systemctl reload
nginx'") | sort -u | crontab -

echo "=== SSL Setup Completed Successfully ==="
